  const quizText = Q => (Q || []).map(q => 'Q: ' + q.q + '\n' + q.choices.map((c, i) => (i === q.answer ? '* ' : '- ') + c).join('\n') + (q.why ? '\nWhy: ' + q.why : '')).join('\n\n');
  const quizParse = t => String(t || '').split(/\n\s*\n/).map(b => { const L = b.split('\n').map(x => x.trim()).filter(Boolean); const q = { q: '', choices: [], answer: 0, why: '' };

        await new Promise((ok, no) => { const x = new XMLHttpRequest(); x.open('PUT', u.url); x.setRequestHeader('Content-Type', f.type || 'video/mp4'); x.setRequestHeader('x-upsert', 'true');
          x.upload.onprogress = ev => { if(ev.lengthComputable) $('tpP').textContent = 'Uploading ' + Math.round(ev.loaded / ev.total * 100) + '%'; };

        await call('part_set', { p: { id: part.id, video_path: u.path, dur_sec: dur, newver: !!part.video_path } }); }
