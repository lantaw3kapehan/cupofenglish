  function decorate(){
    const v = document.getElementById('v-schedule'); if(!v) return;
    v.querySelectorAll('.seg').forEach(s => { if(!/By time/.test(s.textContent)) return;
      s.querySelectorAll('button').forEach(b => { const by = /By time/.test(b.textContent) ? 'time' : /By teacher/.test(b.textContent) ? 'teacher' : ''; if(by){ b.classList.toggle('on', SC_BY === by); b.setAttribute('onclick', "SC_BY='" + by + "';try{localStorage.setItem('coe.scBy','" + by + "')}catch(e){};renderSchedule()"); } });
      if(!s.querySelector('.tl97b')) s.insertAdjacentHTML('beforeend', '<button class="tl97b' + (SC_BY === 'line' ? ' on' : '') + '" onclick="SC_BY=\'line\';try{localStorage.setItem(\'coe.scBy\',\'line\')}catch(e){};renderSchedule()">Timeline</button>'); });
    if(!v.querySelector('.vw03')){ const top = v.querySelector('.s2-top'); if(top) top.insertAdjacentHTML('afterend', '<div class="vw03" style="display:flex;align-items:center;gap:10px;margin:10px 0 2px;flex-wrap:wrap"><span style="font-size:12px;font-weight:600;letter-spacing:.06em;text-transform:uppercase;color:#6F6B85">View</span><div style="display:inline-flex;gap:4px;background:#F1F0F6;border-radius:12px;padding:3px">' +
      [['time', 'List'], ['line', 'Timeline'], ['month', 'Month']].map(([k, l]) => '<button style="border:0;border-radius:9px;padding:8px 16px;min-height:38px;font-weight:600;font-size:14px;font-family:inherit;cursor:pointer;' + (k === SC_BY || (k === 'time' && SC_BY === 'teacher') ? 'background:#fff;color:#2B2253;box-shadow:0 1px 3px rgba(43,34,83,.14)' : 'background:transparent;color:#4A4566') + '" onclick="SC_BY=\'' + k + '\';try{localStorage.setItem(\'coe.scBy\',\'' + k + '\')}catch(e){};renderSchedule();' + (k === 'line' ? 'setTimeout(()=>{const w=document.querySelector(\'#v-schedule .tl97w\');if(w)w.scrollIntoView({behavior:\'smooth\',block:\'start\'})},60)' : '') + '">' + l + '</button>').join('') + '</div><span style="font-size:12.5px;color:#6F6B85">Timeline = every teacher\'s day side by side · Month = the whole month at a glance</span></div>'); }
    v.querySelectorAll('.vw03 button').forEach(b => { const k = b.textContent === 'List' ? 'time' : b.textContent === 'Timeline' ? 'line' : 'month', on = k === SC_BY || (k === 'time' && SC_BY === 'teacher');
      b.style.background = on ? '#fff' : 'transparent'; b.style.color = on ? '#2B2253' : '#4A4566'; b.style.boxShadow = on ? '0 1px 3px rgba(43,34,83,.14)' : 'none'; });
    if(SC_BY !== 'month'){ v.querySelectorAll('.mo04w').forEach(x => x.remove()); v.querySelectorAll('[data-h04]').forEach(x => { x.style.display = ''; delete x.dataset.h04; }); }
    if(SC_BY !== 'line'){ v.querySelectorAll('.tl97w').forEach(x => x.remove()); v.querySelectorAll('[data-h97]').forEach(x => { x.style.display = ''; delete x.dataset.h97; }); }
    if(SC_BY === 'month'){ if(window.__mo04Paint) window.__mo04Paint(v); return; }
    if(SC_BY !== 'line') return;
    const k = v.querySelector('.bkl'); let list = k && k.nextElementSibling; while(list && list.classList.contains('tl97w')) list = list.nextElementSibling; if(!list) return;
    let w = v.querySelector('.tl97w'); const isNew = !w; if(!w){ w = document.createElement('div'); w.className = 'tl97w'; list.insertAdjacentElement('beforebegin', w); }
    w.innerHTML = html(); list.style.display = 'none'; list.dataset.h97 = 1;
    if(isNew) try{ const sc = w.querySelector('.tl97'), nw = w.querySelector('.t7nw'); if(sc && nw){ requestAnimationFrame(() => { const x = nw.getBoundingClientRect().left - sc.getBoundingClientRect().left + sc.scrollLeft; sc.scrollLeft = Math.max(0, x - 160 - (sc.clientWidth - 160) / 3); }); } }catch(e){}
    const hid = v.querySelector('.sc-hid'); if(hid){ hid.style.display = 'none'; hid.dataset.h97 = 1; }
  }
