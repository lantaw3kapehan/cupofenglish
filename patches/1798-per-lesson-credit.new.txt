function lv1795plan(x){   /* 1798: each booked lesson gets its own notice and its own outcome */
  const cls = lv1795cls(x), at = lv1795at(x.at), planned = LV1795_PLANNED.test(x.type || ''), proof = (x.proof || []).length, age = (Date.now() - at) / 36e5;
  return cls.map(c => { const h = (c.t - at) / 36e5; let k = 'none', why = '';
    if(h < 4){ if(proof){ k = 'Excused'; why = 'less than 4 h · with proof'; } else if(!planned && age < 48){ k = 'wait'; why = 'less than 4 h · proof due'; } else { k = 'Unexcused'; why = 'less than 4 h · no proof'; } }
    else if(planned){ if(h < 168){ k = 'Excused'; why = 'planned · less than 7 days'; } else why = 'planned · 7 days or more'; }
    else if(proof){ why = '4 h or more · with proof'; }
    else if(age < 48){ k = 'wait'; why = '4 h or more · proof due'; }
    else { k = 'Unexcused'; why = 'no proof after 48 h'; }
    return Object.assign({}, c, { h: h, k: k, why: why }); }); }

  if(k === 'none'){ try{ localStorage.setItem('lv1795:' + id, JSON.stringify({ k: 'No charge', at: stamp })); }catch(e){} toast('✓ No charge recorded for ' + tName(x.teacher)); lvSheet(id); return; }
  const plan = lv1795plan(x).map(p => k === 'plan' ? p : Object.assign({}, p, { k: 'Unexcused' })).filter(p => p.k === 'Excused' || p.k === 'Unexcused');
  if(!plan.length) return toast('Nothing to charge');
  const total = plan.reduce((a, p) => a + LV1795_RATE[p.k][tier], 0);
  if(!confirm(plan.map(p => p.md + ' ' + p.hm + ' ' + p.name + ' · ' + p.k + ' ₱' + LV1795_RATE[p.k][tier]).join('\n') + '\n\nTotal ₱' + total.toLocaleString() + ' for ' + tName(x.teacher) + ' · one Teacher Class Cancellation case per lesson?')) return;
  toast('Saving ' + plan.length + ' case' + (plan.length === 1 ? '' : 's') + '…'); const ids = [];

      action: 'Notice of unavailability ' + id + ' · ' + lv1795h(c.h) + ' notice · lesson ' + c.md + ' ' + c.hm + ' · Clause ' + (c.h < 4 ? '2.9' : '2.10 / 4.1') }); if(r && r.ok) ids.push(r.caseId); }catch(e){} }
  try{ localStorage.setItem('lv1795:' + id, JSON.stringify({ k: (k === 'plan' ? 'Suggested' : 'All unexcused') + ' · ₱' + total.toLocaleString(), at: stamp, ids: ids.join(', ') })); }catch(e){}
  toast(ids.length === plan.length ? '✓ ' + ids.length + ' case' + (ids.length === 1 ? '' : 's') + ' saved · ' + ids.join(', ') : ids.length + ' of ' + plan.length + ' saved · check Actions'); lvSheet(id); try{ AC = null; }catch(e){} }
