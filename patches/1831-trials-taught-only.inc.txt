  const near = (a, b) => Math.abs(Date.parse(a + 'T00:00:00Z') - Date.parse(b + 'T00:00:00Z')) <= 864e5;
  function apply(){ if(!T || typeof D === 'undefined' || !D || !Array.isArray(D.special)) return; const t0 = today(), lo = new Date(Date.now() + 8 * 3600e3 - 44 * 864e5).toISOString().slice(0, 10);
    const keep = D.special.filter(x => { if(x.type !== 'Trial' || !x.code || !x.date) return true; const d = String(x.date).slice(0, 10); if(d >= t0 || d < lo) return true;
      return T.some(c => c.code === String(x.code).trim().toUpperCase() && near(c.day, d)); });
    if(keep.length !== D.special.length){ D.special = keep; try{ render(); }catch(e){} } }
  async function check(){ try{ if(typeof D === 'undefined' || !D || D.mgr || !TOKEN || !(D.special || []).some(x => x.type === 'Trial')) return;

      BUSY = 0; if(r && r.ok && Array.isArray(r.list)){ T = r.list; AT = Date.now(); apply(); } }catch(e){ BUSY = 0; } }
  const _r = window.render; if(_r) window.render = function(){ const o = _r.apply(this, arguments); setTimeout(check, 0); return o; };
})();
