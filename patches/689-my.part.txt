  function ladderHtml(d, who, R){ const L = statusList(d, who, R.ix), G = [];
    L.forEach(x => { const g = G[G.length - 1]; if(g && g.gate === x.t.gate) g.items.push(x); else G.push({ gate: x.t.gate, gi: x.gi, items: [x] }); });
    return '<div class="tx-lad">' + G.slice().reverse().map(g => { const st = g.gi <= R.ix ? 'ok' : g.gi === R.ix + 1 ? 'nx' : 'lk';
      const sub = g.gi === R.ix ? 'you are here' : g.gi === R.ix + 1 && R.next ? (Math.round((+R.life || 0) * 10) / 10).toLocaleString() + ' / ' + Math.round(R.next.units != null ? +R.next.units : (+R.life || 0) + (+R.next.need || 0)).toLocaleString() + ' units' : '';
      return '<div class="tx-rung ' + st + '"><div class="tx-rl">' + e(RS[g.gate] || g.gate) + (st === 'ok' ? ' ✓' : st === 'nx' ? ' · next' : '') + (sub ? '<small>' + e(sub) + '</small>' : '') + '</div><div class="tx-bar" style="background:' + (st === 'lk' ? 'var(--line)' : RC[g.gate] || 'var(--teal)') + '"></div><div class="tx-pills">'
        + g.items.map(x => { const cls = x.st === 'done' ? 'done' : x.st === 'next' ? 'next' : x.st === 'open' ? 'open' : 'lock';
          const lab = (cls === 'done' ? '✓ ' : cls === 'next' ? '▶ ' : '') + e(x.t.title) + (cls === 'open' && x.gi <= R.ix && x.t.id !== 'way' ? ' <small>not recorded yet</small>' : '');
          return '<button type="button" class="tx-pill ' + cls + '" onclick="trOpen(\'' + x.t.id + '\')">' + (cls === 'lock' ? LOCK : '') + '<span>' + lab + '</span></button>'; }).join('') + '</div></div>'; }).join('') + '</div>'; }
